# Contains common functions used by all cmake files, such as finding various libraries
# or checking for the presence of specific system features


function (findOpenMP)
# unset OpenMP_CXX_FLAGS. If OpenMP loading fails once, it will be set to NOT_FOUND
# and happily cached, thus messing up the compilation flags forever
unset(OpenMP_C_FLAGS CACHE)
unset(OpenMP_CXX_FLAGS CACHE)
if(CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang")
  if(NOT OpenMP_ROOT)
    set(OpenMP_ROOT "$ENV{OpenMP_ROOT}")
  endif()
  if(NOT OpenMP_ROOT)
    execute_process(
        COMMAND brew --prefix libomp
        OUTPUT_VARIABLE OpenMP_ROOT
        OUTPUT_STRIP_TRAILING_WHITESPACE
    )
  endif()
  set(OpenMP_C_FLAGS "-Xclang -fopenmp -I${OpenMP_ROOT}/include")
  set(OpenMP_CXX_FLAGS "-Xclang -fopenmp -I${OpenMP_ROOT}/include")
  set(OpenMP_C_LIB_NAMES libomp)
  set(OpenMP_CXX_LIB_NAMES libomp)
  set(OpenMP_libomp_LIBRARY ${OpenMP_ROOT}/lib/libomp.dylib)
endif()
find_package(OpenMP REQUIRED)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} ${OpenMP_C_FLAGS}" PARENT_SCOPE)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${OpenMP_CXX_FLAGS}" PARENT_SCOPE)
set(OpenMP_CXX_FLAGS "${OpenMP_CXX_FLAGS}" PARENT_SCOPE)
set(OpenMP_CXX_LIBRARIES "${OpenMP_CXX_LIBRARIES}" PARENT_SCOPE)
set(OpenMP_CXX_VERSION "${OpenMP_CXX_VERSION}" PARENT_SCOPE)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU" OR NOT ${CMAKE_SYSTEM_NAME} MATCHES "Darwin")
set(OpenMP_CXX_LIBRARIES ${OpenMP_CXX_LIBRARIES} -latomic PARENT_SCOPE)
endif()
endfunction()

# check for std::filesystem::temp_directory_path
function (checkFileSystem)
include(CheckCXXSourceRuns)
set(CMAKE_REQUIRED_FLAGS " -std=c++17")
check_cxx_source_runs("
    #include <iostream>
    #include <filesystem>
    int main() {
        std::cout << std::filesystem::temp_directory_path();
        return 0;
    }
" CPPNOFS)
if(NOT CPPNOFS)
  set(CMAKE_REQUIRED_FLAGS " -std=c++17")
  set(CMAKE_REQUIRED_LIBRARIES "c++fs")
  check_cxx_source_runs("
      #include <iostream>
      #include <filesystem>
      int main() {
          std::cout << std::filesystem::temp_directory_path();
          return 0;
      }
  " CPPFS)
  unset(CMAKE_REQUIRED_FLAGS)
  unset(CMAKE_REQUIRED_LIBRARIES)
  if(CPPFS)
    set(FILESYSTEM_LIBRARY -lc++fs PARENT_SCOPE)
  else()
    set(CMAKE_REQUIRED_FLAGS " -std=c++17")
    set(CMAKE_REQUIRED_LIBRARIES "stdc++fs")
    check_cxx_source_runs("
        #include <iostream>
        #include <filesystem>
        int main() {
            std::cout << std::filesystem::temp_directory_path();
            return 0;
        }
    " STDCPPFS)
    unset(CMAKE_REQUIRED_FLAGS)
    unset(CMAKE_REQUIRED_LIBRARIES)
    if(STDCPPFS)
      set(FILESYSTEM_LIBRARY -lstdc++fs PARENT_SCOPE)
    else()
      message(FATAL_ERROR "std::filesystem not found")
    endif()
  endif()
endif()
endfunction()

function (buildLogging)
  set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
  set(SPDLOG_FMT_EXTERNAL ON)
  add_subdirectory(${CMAKE_CURRENT_FUNCTION_LIST_DIR}/external-libraries/fmt fmt)
  add_subdirectory(${CMAKE_CURRENT_FUNCTION_LIST_DIR}/external-libraries/spdlog spdlog)
endfunction()

# Share dependency targets between the full build and standalone common-library builds.
function (buildCommonDependencies)
  if(TARGET metagraph-common-dependencies)
    return()
  endif()
  set(EXTERNAL_LIB_DIR ${CMAKE_CURRENT_FUNCTION_LIST_DIR}/external-libraries)
  set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
  add_subdirectory(${EXTERNAL_LIB_DIR}/hopscotch-map hopscotch-map EXCLUDE_FROM_ALL)
  add_subdirectory(${EXTERNAL_LIB_DIR}/ordered-map ordered-map EXCLUDE_FROM_ALL)
  add_subdirectory(${EXTERNAL_LIB_DIR}/DYNAMIC DYNAMIC EXCLUDE_FROM_ALL)
  add_subdirectory(${EXTERNAL_LIB_DIR}/sdsl-lite sdsl-lite EXCLUDE_FROM_ALL)

  # These dependencies do not provide CMake targets.
  add_library(ips4o INTERFACE)
  target_include_directories(ips4o INTERFACE ${EXTERNAL_LIB_DIR}/ips4o)
  add_library(simde INTERFACE)
  target_include_directories(simde SYSTEM INTERFACE ${EXTERNAL_LIB_DIR}/simde-no-tests)

  add_library(metagraph-common-dependencies INTERFACE)
  target_link_libraries(metagraph-common-dependencies INTERFACE
    tsl::hopscotch_map tsl::ordered_map XXSDS_DYNAMIC sdsl-lite::sdsl-lite ips4o simde)
endfunction()

function (findFolly)
  set(FOLLY_SOURCE_DIR ${CMAKE_CURRENT_FUNCTION_LIST_DIR}/external-libraries/folly)
  unset(FOLLY_FOUND PARENT_SCOPE)
  set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} ${CMAKE_CURRENT_FUNCTION_LIST_DIR})
  find_package(Jemalloc)

  if(Jemalloc_FOUND)
    if(${JEMALLOC_VERSION} VERSION_LESS 4.0)
      message(WARNING "jemalloc version less than 4.0")
      unset(Jemalloc_FOUND)
    endif()
  endif()
  set(Jemalloc_FOUND ${Jemalloc_FOUND} PARENT_SCOPE)

  message(STATUS "Configuring Folly")
  set(_folly_old_req_includes "${CMAKE_REQUIRED_INCLUDES}")
  if(JEMALLOC_INCLUDE_DIRS)
    list(APPEND CMAKE_REQUIRED_INCLUDES ${JEMALLOC_INCLUDE_DIRS})
  endif()
  include(${FOLLY_SOURCE_DIR}/CMake/FollyConfigChecks.cmake)
  set(CMAKE_REQUIRED_INCLUDES "${_folly_old_req_includes}")
  configure_file(
    ${FOLLY_SOURCE_DIR}/CMake/folly-config.h.cmake
    ${PROJECT_BINARY_DIR}/external-libraries/folly/include/folly/folly-config.h
  )
  # Folly's component targets also pull in unrelated external libraries.
  # Build only the implementation needed by fbvector and small_vector.
  add_library(folly STATIC
    ${FOLLY_SOURCE_DIR}/folly/lang/Exception.cpp
    ${FOLLY_SOURCE_DIR}/folly/lang/SafeAssert.cpp
    ${FOLLY_SOURCE_DIR}/folly/lang/ToAscii.cpp
    ${FOLLY_SOURCE_DIR}/folly/ScopeGuard.cpp
    ${FOLLY_SOURCE_DIR}/folly/memory/detail/MallocImpl.cpp
    ${FOLLY_SOURCE_DIR}/folly/memory/Malloc.cpp
    ${FOLLY_SOURCE_DIR}/folly/memory/SanitizeLeak.cpp
  )
  target_compile_features(folly PUBLIC cxx_std_20)
  if(Jemalloc_FOUND)
    target_compile_definitions(folly PRIVATE USE_JEMALLOC JEMALLOC_NO_DEMANGLE)
  endif()
  target_include_directories(folly SYSTEM
    PRIVATE ${JEMALLOC_INCLUDE_DIRS}
    PUBLIC
      ${FOLLY_SOURCE_DIR}
      ${PROJECT_BINARY_DIR}/external-libraries/folly/include
  )
  target_link_libraries(folly PUBLIC fmt::fmt PRIVATE Boost::headers ${JEMALLOC_LIBRARIES})
  set(FOLLY_FOUND TRUE PARENT_SCOPE)
endfunction()


function (buildProgressBar EXTERNAL_LIB_DIR libName)
set(${libName} progress_bar PARENT_SCOPE)
if(TARGET progress_bar)
  return()
endif()
add_library(progress_bar STATIC
    ${EXTERNAL_LIB_DIR}/cpp_progress_bar/progress_bar.cpp
)
target_include_directories(progress_bar PUBLIC ${EXTERNAL_LIB_DIR}/cpp_progress_bar)
endfunction()


function (buildMersenneTwister EXTERNAL_LIB_DIR libName)
  set(${libName} mersenne_twister PARENT_SCOPE)
  if(TARGET mersenne_twister)
    return()
  endif()
  add_library(mersenne_twister STATIC
    ${EXTERNAL_LIB_DIR}/rollinghashcpp/mersennetwister.cpp
  )
  target_include_directories(mersenne_twister PUBLIC ${EXTERNAL_LIB_DIR}/rollinghashcpp)
  # Legacy arithmetic between enum constants is valid but deprecated in C++20.
  target_compile_options(mersenne_twister PRIVATE
    $<$<CXX_COMPILER_ID:GNU>:-Wno-error=deprecated-enum-enum-conversion>
    $<$<CXX_COMPILER_ID:Clang,AppleClang>:-Wno-error=deprecated-anon-enum-enum-conversion>
  )
endfunction()


function (findGTest)
if(TARGET gtest)
  return()
endif()
add_subdirectory(${EXTERNAL_LIB_DIR}/googletest googletest EXCLUDE_FROM_ALL)
target_compile_options(gtest_main PRIVATE -w)
target_compile_options(gtest PRIVATE -w)
set(GTEST_BOTH_LIBRARIES gtest gtest_main PARENT_SCOPE)
endfunction()

# Enables using ccache in order to speed up compilation. Make sure you
# brew install ccache to benefit from it
function (enableCCache)
find_program(CCACHE_FOUND ccache)
if(CCACHE_FOUND)
    # Use modern CMAKE_*_COMPILER_LAUNCHER which propagates better to external projects
    set(CMAKE_CXX_COMPILER_LAUNCHER "${CCACHE_FOUND}" PARENT_SCOPE)
    set(CMAKE_C_COMPILER_LAUNCHER "${CCACHE_FOUND}" PARENT_SCOPE)
else()
    message(STATUS "ccache not found. Run 'brew install ccache' to speed up re-compilation")
endif(CCACHE_FOUND)
endfunction()
